38def load_material_catalog() -> dict[str, Any]: 

39 """Load the material catalog JSON from assets with safe fallbacks.""" 

40 try: 

41 with ASSET_CATALOG_PATH.open("r", encoding="utf-8") as handle: 

42 data = json.load(handle) 

43 except (OSError, json.JSONDecodeError): 

44 return { 

45 "materials": list(_FALLBACK_MATERIALS), 

46 "sellmeier_reference": dict(_FALLBACK_SELLMEIER_REFERENCE), 

47 } 

48 

49 materials = data.get("materials", _FALLBACK_MATERIALS) 

50 if not isinstance(materials, list): 

51 materials = list(_FALLBACK_MATERIALS) 

52 

53 sellmeier_reference = data.get("sellmeier_reference", [_FALLBACK_SELLMEIER_REFERENCE]) 

54 if isinstance(sellmeier_reference, dict): 

55 sellmeier_reference = [sellmeier_reference] 

56 if not isinstance(sellmeier_reference, list): 

57 sellmeier_reference = [dict(_FALLBACK_SELLMEIER_REFERENCE)] 

58 

59 return { 

60 "materials": materials, 

61 "sellmeier_reference": sellmeier_reference, 

62 }

65def material_dropdown_options() -> list[dict[str, str]]: 

66 """Build dropdown options from the JSON stock material catalog.""" 

67 catalog = load_material_catalog() 

68 options = [] 

69 

70 for name in catalog["materials"]: 

71 if not isinstance(name, str): 

72 continue 

73 normalized = name.strip() 

74 if not normalized: 

75 continue 

76 

77 options.append({"label": _to_display_name(normalized), "value": normalized}) 

78

82def _to_display_name(name: str) -> str: 

83 """Convert canonical material keys into user-facing labels.""" 

84 if "_" in name: 

85 return " ".join(part.upper() if part.isupper() else part.capitalize() for part in name.split("_")) 

86 if any(char.islower() for char in name) and any(char.isupper() for char in name): 

87 return name 

88 if name.isupper() or any(char.isdigit() for char in name): 

89 return name 

90 return name.capitalize()
