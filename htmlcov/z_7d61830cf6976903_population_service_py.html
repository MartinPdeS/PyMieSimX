30def build_population_distribution(*, distribution="lognormal", diameter_nm=500, width=1.2, 

31 minimum_nm=100, maximum_nm=1000, sampling=128): 

32 """Build native number fractions; Gaussian width is in nm, lognormal width is GSD.""" 

33 if distribution == "monodisperse": 

34 return ParticleSizeDistribution.monodisperse(_scalar(diameter_nm, "Diameter") * ureg.nm) 

35 count = _scalar(sampling, "Distribution sampling") 

36 if not count.is_integer() or not 16 <= count <= MAX_POPULATION_NODES: 

37 raise ValueError(f"Distribution sampling must be an integer from 16 to {MAX_POPULATION_NODES}.") 

38 if distribution == "lognormal": 

39 median = _scalar(diameter_nm, "Median diameter") 

40 gsd = _scalar(width, "Geometric standard deviation") 

41 if not 1 < gsd <= 3: 

42 raise ValueError("Geometric standard deviation must be greater than 1 and at most 3.") 

43 return ParticleSizeDistribution.lognormal(median * ureg.nm, gsd, sampling=int(count)) 

44 if distribution not in {"gaussian", "uniform"}: 

45 raise ValueError("Select monodisperse, Gaussian, lognormal, or uniform.") 

46 lower = _scalar(minimum_nm, "Minimum diameter") 

47 upper = _scalar(maximum_nm, "Maximum diameter") 

48 if upper <= lower: 

49 raise ValueError("Maximum diameter must exceed minimum diameter.") 

50 if distribution == "uniform": 

51 return ParticleSizeDistribution.uniform(lower * ureg.nm, upper * ureg.nm, sampling=int(count)) 

52 mean = _scalar(diameter_nm, "Gaussian mean diameter") 

53 std = _scalar(width, "Gaussian standard deviation") 

54 if not lower <= mean <= upper: 

55 raise ValueError("Gaussian mean must lie within the diameter bounds.") 

56 return ParticleSizeDistribution.truncated_normal( 

57 mean * ureg.nm, std * ureg.nm, minimum_diameter=lower * ureg.nm, 

58 maximum_diameter=upper * ureg.nm, sampling=int(count),

62def parse_population_material(value, *, medium=False): 

63 parsed = parse_material_values(value, medium=medium) 

64 if parsed is None or isinstance(parsed, (list, tuple, np.ndarray)): 

65 raise ValueError("Material and medium must each be one refractive index or material name.") 

66 if isinstance(parsed, (int, float, complex)): 

67 index = complex(parsed) 

68 if not np.isfinite(index) or index.real <= 0 or index.imag < 0 or (medium and index.imag != 0): 

69 raise ValueError("Use a positive real medium index and a particle index n+ik with n > 0 and k >= 0.") 

70 return index.real if medium else index 

71 return parsed

172def export_population_to_csv(result): 

173 """Export results, units, definitions, input provenance, and discrete number fractions.""" 

174 output = StringIO() 

175 writer = csv.writer(output, lineterminator="\n") 

176 writer.writerow(["parameter", "value", "unit", "definition"]) 

177 for row in result["results"]: 

178 writer.writerow([row[key] for key in ("parameter", "value", "unit", "definition")]) 

179 writer.writerow([]) 

180 writer.writerow(["approximation", result["approximation"]]) 

181 writer.writerow(["input", "value"]) 

182 writer.writerows(result["inputs"].items()) 

183 writer.writerow([]) 

184 writer.writerow(["diameter_nm", "number_fraction"]) 

185 writer.writerows(zip(result["distribution"]["diameters_nm"], result["distribution"]["number_fractions"])) 

186 return output.getvalue()
